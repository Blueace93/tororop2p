// === Mejoras de rendimiento para conexiones lentas === // Mantiene la apariencia y las animaciones del login/entrada intactas. // Solo reduce la carga inicial y el uso de recursos no esenciales. const PERF = { turnstileLoaded: false, ocrLoaded: false, turnstileReady: null, ocrReady: null, loadQueue: [] }; function loadScript(src, attrs = {}) { return new Promise((resolve, reject) => { const existing = document.querySelector(`script[src="${src}"]`); if (existing) { if (existing.dataset.loaded === '1') return resolve(existing); existing.addEventListener('load', () => resolve(existing), { once: true }); existing.addEventListener('error', () => reject(new Error(`Error cargando ${src}`)), { once: true }); return; } const s = document.createElement('script'); s.src = src; Object.entries(attrs).forEach(([k, v]) => s.setAttribute(k, v)); s.async = true; s.addEventListener('load', () => { s.dataset.loaded = '1'; resolve(s); }, { once: true }); s.addEventListener('error', () => reject(new Error(`Error cargando ${src}`)), { once: true }); document.head.appendChild(s); }); } function loadTurnstile() { if (PERF.turnstileLoaded) return Promise.resolve(PERF.turnstileReady); PERF.turnstileLoaded = true; const src = 'https://challenges.cloudflare.com/turnstile/v0/api.js?render=explicit'; return loadScript(src, { async: 'true', defer: 'true' }) .then(() => { PERF.turnstileReady = window.turnstile; return window.turnstile; }) .catch(err => { console.warn('[perf] Turnstile no pudo cargarse:', err); PERF.turnstileReady = null; return null; }); } function ensureTesseract() { if (PERF.ocrLoaded) return Promise.resolve(PERF.ocrReady); PERF.ocrLoaded = true; const scripts = [ 'tesseract.min.js', 'worker.min.js' ]; return Promise.allSettled(scripts.map(src => loadScript(src, { async: 'true' }))) .then(() => { PERF.ocrReady = window.Tesseract || null; return PERF.ocrReady; }) .catch(() => { PERF.ocrReady = null; return null; }); } // Carga diferida: solo cuando el usuario entra a un estado donde se necesita // No bloquea la pantalla inicial ni la animación de entrada. function lazyLoadForFeature(feature) { if (feature === 'turnstile') return loadTurnstile(); if (feature === 'ocr') return ensureTesseract(); return Promise.resolve(null); } // Aumenta el rendimiento en conexiones lentas usando un timeout más realista // y evitando reintentos agresivos en lectura. function withSlowNetworkTimeout(promiseFactory, ms = 8000) { return new Promise((resolve, reject) => { const timer = setTimeout(() => reject(new Error('slow-network-timeout')), ms); Promise.resolve() .then(promiseFactory) .then(val => { clearTimeout(timer); resolve(val); }) .catch(err => { clearTimeout(timer); reject(err); }); }); } // Evita ejecutar inicializadores pesados justo en el arranque. function scheduleIdleWork(fn, delay = 150) { if ('requestIdleCallback' in window) { requestIdleCallback(() => setTimeout(fn, delay)); } else { setTimeout(fn, delay); } } // Hook: se dispara cuando el usuario va a una vista que necesita Turnstile. // Si no se usa, nunca se carga. document.addEventListener('click', e => { const target = e.target.closest && e.target.closest('[data-needs-turnstile]'); if (!target) return; lazyLoadForFeature('turnstile'); }); // Hook: si el usuario intenta subir un documento o inspeccionar imagen con OCR, // se carga solo entonces. document.addEventListener('click', e => { const target = e.target.closest && e.target.closest('[data-needs-ocr]'); if (!target) return; lazyLoadForFeature('ocr'); }); // Hook de arranque: no bloquea la pantalla inicial; solo carga en idle. window.addEventListener('load', () => { scheduleIdleWork(() => { // No cargamos nada pesado aquí, solo dejamos la app lista para entrar sin penalizar. // Turnstile/OCR se carga al entrar a la vista correspondiente. }, 200); }); // Ajuste de reintentos para peticiones de lectura en conexiones pobres. const slowNetFetch = (url, opt = {}) => { const method = (opt.method || 'GET').toUpperCase(); const readOnly = method === 'GET' || method === 'HEAD'; if (!readOnly) return fetch(url, opt); return new Promise((resolve, reject) => { let attempts = 0; const run = () => { attempts += 1; const controller = new AbortController(); const timeout = setTimeout(() => controller.abort(), 8000); fetch(url, { ...opt, signal: controller.signal }) .then(res => { clearTimeout(timeout); if (res.ok) return resolve(res); if (attempts < 2) return setTimeout(run, 500); resolve(res); }) .catch(err => { clearTimeout(timeout); if (attempts < 2 && !controller.signal.aborted) return setTimeout(run, 600); reject(err); }); }; run(); }); }; // Reemplazar fetch global solo si ya existe y no rompe la plataforma. if (window.fetch && !window.__TORORO_SLOWNET_FETCH__) { window.__TORORO_SLOWNET_FETCH__ = window.fetch; window.fetch = (input, init) => slowNetFetch(input, init || {}); }
